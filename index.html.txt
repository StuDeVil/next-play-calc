  <script>
    function calculateProfit() {
      const gamePrice = parseFloat(document.getElementById('game-price').value) || 0;
      const unitsSold = parseInt(document.getElementById('units-sold').value) || 0;
      const publisherFee = parseFloat(document.getElementById('publisher-fee').value) / 100 || 0;
      const platformFee = parseFloat(document.getElementById('platform-fee').value) / 100 || 0;
      const engineRoyalty = parseFloat(document.getElementById('engine-royalty').value) / 100 || 0;
      const marketingCosts = parseFloat(document.getElementById('marketing-costs').value) || 0;
      const devCosts = parseFloat(document.getElementById('dev-costs').value) || 0;
      const taxRate = parseFloat(document.getElementById('tax-rate').value) / 100 || 0;

      const grossRevenue = gamePrice * unitsSold;
      const totalFees = grossRevenue * (publisherFee + platformFee + engineRoyalty);
      const netRevenue = grossRevenue - totalFees - (grossRevenue * taxRate);
      const profitMargin = ((netRevenue - marketingCosts - devCosts) / grossRevenue) * 100;
      const breakeven = (marketingCosts + devCosts) / gamePrice;

      document.getElementById('gross-revenue').textContent = grossRevenue.toFixed(2);
      document.getElementById('total-fees').textContent = totalFees.toFixed(2);
      document.getElementById('net-revenue').textContent = netRevenue.toFixed(2);
      document.getElementById('profit-margin').textContent = profitMargin.toFixed(2) + '%';
      document.getElementById('breakeven').textContent = Math.ceil(breakeven) + ' units';
    }
  </script>
